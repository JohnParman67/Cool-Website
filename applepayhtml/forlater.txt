    <script>
        (function() {
            let count = 0; // Initialize counter
            const img = document.getElementById('clickImage');
            const counterDisplay = document.getElementById('counter');

            // Ensure image loads before attaching event
            img.addEventListener('load', () => {
                console.log("Image loaded successfully.");
            });

            img.addEventListener('error', () => {
                console.error("Error: 'something.png' not found.");
                counterDisplay.textContent = "Image not found.";
            });

            // Increment counter on click
            img.addEventListener('click', () => {
                count++;
                counterDisplay.textContent = `Robux: ${count}`;
            });
        })();
    </script>
